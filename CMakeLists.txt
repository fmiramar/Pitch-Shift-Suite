cmake_minimum_required(VERSION 3.20)
project(PitchShiftSuite VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(PITCH_BUILD_SC "Build SuperCollider UGen modules" OFF)
option(PITCH_BUILD_SIGNALSMITH "Build A01 SpectralHybridShift and its smoke test" OFF)
option(PITCH_BUILD_RUBBERBAND "Build optional GPL-2.0-or-later Rubber Band module" OFF)
option(PITCH_BUILD_OLA "Build A10 granular OLA pitch shifter" OFF)
option(PITCH_BUILD_WSOLA "Build A11 waveform-similarity OLA pitch shifter" OFF)
option(PITCH_BUILD_LOW_LATENCY "Build A16 low-latency OCEAN spectral shifter" OFF)
option(PITCH_BUILD_SINUSOIDAL "Build A14 sinusoidal plus stochastic residual engine" OFF)
option(PITCH_BUILD_MULTIRATE "Build A13 Terrarium multirate octave variant" OFF)
option(PITCH_BUILD_PHASE_HALF "Build A12 experimental subband phase-halving engine" OFF)
option(PITCH_BUILD_ERB_SSB "Build A06 ERB single-sideband octave reference" OFF)
option(PITCH_BUILD_ERB_PHASE "Build A05 ERB phase-scaling octave doubler" OFF)
option(PITCH_BUILD_OCEAN "Build A08 diagnostic OCEAN octave shifter" OFF)
option(PITCH_BUILD_PV_INTEGER "Build A07 integer phase-vocoder octave doubler" OFF)
option(PITCH_BUILD_PHASE_LOCKED "Build A09 phase-locked phase vocoder" OFF)
option(PITCH_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(SC_SOURCE_DIR "" CACHE PATH "SuperCollider source tree matching the target server")
add_library(pitch_core INTERFACE)
target_include_directories(pitch_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(pitch_core INTERFACE cxx_std_17)
if(PITCH_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizers require Clang or GCC")
  endif()
  target_compile_options(pitch_core INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(pitch_core INTERFACE -fsanitize=address,undefined)
endif()
if(PITCH_BUILD_SC AND NOT EXISTS "${SC_SOURCE_DIR}/include/plugin_interface/SC_PlugIn.h")
  message(FATAL_ERROR "Set SC_SOURCE_DIR to the matching SuperCollider source checkout")
endif()

  target_link_libraries(pitch_signalsmith INTERFACE pitch_core)

    add_executable(signalsmith_smoke tests/signalsmith_smoke.cpp)
    target_link_libraries(signalsmith_smoke PRIVATE pitch_signalsmith)
    add_test(NAME signalsmith_smoke COMMAND signalsmith_smoke)
  endif()
  if(PITCH_BUILD_SC)
    add_library(SpectralHybridShift MODULE src/sc/SpectralHybridShift.cpp)
    target_link_libraries(SpectralHybridShift PRIVATE pitch_signalsmith)
    target_include_directories(SpectralHybridShift PRIVATE ${SC_SOURCE_DIR}/include/plugin_interface
      ${SC_SOURCE_DIR}/include/common ${SC_SOURCE_DIR}/common)
    set_target_properties(SpectralHybridShift PROPERTIES PREFIX "")
    if(APPLE OR WIN32)
      set_target_properties(SpectralHybridShift PROPERTIES SUFFIX ".scx")
    endif()
    install(TARGETS SpectralHybridShift LIBRARY DESTINATION SpectralHybridShift COMPONENT SpectralHybridShift
      RUNTIME DESTINATION SpectralHybridShift COMPONENT SpectralHybridShift)
    install(FILES sc/Classes/SpectralHybridShift.sc DESTINATION SpectralHybridShift/Classes COMPONENT SpectralHybridShift)
    install(FILES sc/HelpSource/Classes/SpectralHybridShift.schelp DESTINATION SpectralHybridShift/HelpSource/Classes COMPONENT SpectralHybridShift)
    install(FILES third_party/signalsmith-stretch/LICENSE.txt DESTINATION SpectralHybridShift/licenses RENAME Signalsmith-Stretch.txt COMPONENT SpectralHybridShift)
    install(FILES third_party/signalsmith-linear/LICENSE.txt DESTINATION SpectralHybridShift/licenses RENAME Signalsmith-Linear.txt COMPONENT SpectralHybridShift)
  endif()
endif()
